cmake_minimum_required(VERSION 3.24)
project(iiXmlHtmlConsumer LANGUAGES CXX)
include(CTest)
find_package(iiXml 0.1 CONFIG REQUIRED COMPONENTS HtmlBlock)
find_package(iiHtmlBlock 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_compile_features(consumer PRIVATE cxx_std_23)
target_link_libraries(consumer PRIVATE iiXml::HtmlBlock iiHtmlBlock::iiHtmlBlock)
add_test(NAME iiXml.installed_html COMMAND consumer)
