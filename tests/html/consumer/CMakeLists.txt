cmake_minimum_required(VERSION 3.24)
project(iiXmlHtmlConsumer LANGUAGES CXX)
include(CTest)
if(TEST_LEGACY_PACKAGE)
    # The compatibility name must use its own installation's export even when
    # the consumer disabled lookup of the independent iiXml package name.
    set(CMAKE_DISABLE_FIND_PACKAGE_iiXml TRUE)
    find_package(iiHtmlBlock 0.1 CONFIG REQUIRED)
    get_target_property(_html_target iiHtmlBlock::iiHtmlBlock INTERFACE_LINK_LIBRARIES)
    if(NOT _html_target STREQUAL "iiXml::HtmlBlock")
        message(FATAL_ERROR "Legacy package did not select the integrated HTML component")
    endif()
    set(_target iiHtmlBlock::iiHtmlBlock)
else()
    set(CMAKE_DISABLE_FIND_PACKAGE_iiHtmlBlock TRUE)
    find_package(iiXml 0.1 CONFIG REQUIRED COMPONENTS HtmlBlock)
    get_target_property(_xml_libraries iiXml::iiXml INTERFACE_LINK_LIBRARIES)
    if(TARGET iiHtmlBlock::iiHtmlBlock OR _xml_libraries MATCHES "iiHtmlBlock|iiXml::HtmlBlock")
        message(FATAL_ERROR "The XML library acquired a dependency on the HTML component")
    endif()
    set(_target iiXml::HtmlBlock)
endif()
add_executable(consumer main.cpp)
target_compile_features(consumer PRIVATE cxx_std_23)
target_link_libraries(consumer PRIVATE ${_target})
add_test(NAME iiXml.installed_html COMMAND consumer)
